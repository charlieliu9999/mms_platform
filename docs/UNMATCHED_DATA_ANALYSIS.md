```python
# 在 NLP pipeline 的 pre-process 步骤添加
def split_multiple_items(text):
    """拆分逗号分隔的多项目"""
    if ',' in text and text.count(',') <= 3:
        return [item.strip() for item in text.split(',')]
    return [text]

def normalize_position_phrase(text):
    """规范化 正、侧位 等短语"""
    replacements = {
        '正、侧位': '正侧位',
        '正、斜位': '正斜位',
        '正、侧、斜位': '正侧斜位'
    }
    for old, new in replacements.items():
        text = text.replace(old, new)
    return text
```
